\section{支撑材料文件列表}

\begin{itemize}
  \item \texttt{data/附件1.xlsx}：代表日电价、小区负载与光伏预测功率（题面附件 1）；
  \item \texttt{data/附件2.xlsx}：2025 年逐日实测负载与光伏功率（题面附件 2）；
  \item \texttt{data/附件3.xlsx}：2025 年逐日 0:00、6:00、12:00、18:00 发布的未来 24 小时整点光伏功率预报（题面附件 3）；
  \item \texttt{results/result1.xlsx}：问题一提交结果（计划购电量、充放电量两个工作表）；
  \item \texttt{results/result2.xlsx}：问题二提交结果（计划购电量、充放电量、紧急购电量三个工作表，覆盖 2025.2.1--12.31）；
  \item \texttt{问题四/result4-2.xlsx}：问题四 Q2 分支提交结果（实时电价下重算问题二）；
  \item \texttt{问题四/result4-3.xlsx}：问题四 Q3 分支提交结果（实时电价下重算问题三，另含调整购电量工作表）；
  \item \texttt{问题一/算法/q1\_solve.py}：问题一主程序（线性规划装配、求解、独立校验、写表）；
  \item \texttt{问题一/算法/q1\_analysis.py}：问题一无储能基准、容量／功率敏感性与边际价值；
  \item \texttt{问题二/代码/q2\_main\_algorithm.py}：问题二主程序（预测—计划—执行三层框架自包含实现，运行约 3\,s 校验总费用 1395.6995 万元）；
  \item \texttt{问题三/代码/}：问题三的多时刻预报融合、调整层线性规划与执行段函数（\S\ref{sec:q3-forecast}--\S\ref{sec:q3-adjust} 的实现）；
  \item \texttt{问题三/图片/}：问题三全部结果图（$2\times2$ 费用矩阵、全融合与门控对照、指定日调度），即正文图~\ref{fig:q3-matrix}--图~\ref{fig:q3-month}；
  \item \texttt{问题四/代码/}：问题四的 48 小时滚动时域计划与调整线性规划、尽限执行与全年回放函数（\S\ref{sec:q4-model}--\S\ref{sec:q4-check} 的实现）；
  \item \texttt{问题四/照片/}：问题四全部结果图（电价结构分解、方案阶梯、时域机制、遗憾与终端归一化），即正文图~\ref{fig:q4-struct}--图~\ref{fig:q4-regret}；
  \item \texttt{scripts/q4\_export\_pt.py}：导出问题二预测器的中心预测序列，供问题四链路复现使用；
  \item \texttt{scripts/q2\_dow\_N\_joint.py}：主方案参数选取（36 候选逐月 walk-forward）；
  \item \texttt{scripts/q2\_V\_dow\_N\_profile.py}：主方案月度费用与风险档案；
  \item \texttt{scripts/q2\_finalize\_E.py}：主方案唯一入口（复现总费用、论文三张表、写 \texttt{result2.xlsx}）；
  \item \texttt{results/*.json}：上述各脚本的数值结果存档。
\end{itemize}
