import numpy as np
from scipy.optimize import linprog

T, DT, MULT = 144, 1/6, 5.0                # 区间数 / 步长 h / 紧急购电倍数
E_MIN, E_MAX, P_MAX = 1200.0, 10800.0, 5000.0   # SOC 限 kWh / 充放限 kW
ETA_C = ETA_D = 0.95                       # 充放效率；ISSUES = [0, 6, 12, 18]

def pv_energy(D, fc, d, issue):            # 附件3整点功率点值 -> 区间电量(kWh)
    xp = np.arange(issue, 25.0)            # 时刻 tau..24（首端点 = tau 实测功率）
    fp = np.concatenate([[D['pv'][d, :issue*6].mean() / DT if issue
                          else 0.0], fc[:24 - issue]])
    tt = np.arange(T) / 6.0;  ok = tt >= issue
    pa, pb = np.interp(tt[ok], xp, fp), np.interp(tt[ok] + DT, xp, fp)
    out = np.zeros(T);  out[ok] = 0.5 * (pa + pb) * DT
    return out                             # 相邻整点线性插值 + 区间积分

def est_kappa(d, D, pvfc, base_fn, K=30, MIN_SMP=7):
    """用过去 K 天数据在 0:00 通道上做因果滚动回归，得 kappa0[3]。"""
    days = list(range(max(30, d - K), d))  # 只用 d-1 及以前，严格因果
    if len(days) < MIN_SMP:
        return np.zeros(3)                 # 样本不足回退 0
    k0 = np.zeros(3)
    for g, (a, b) in enumerate([(0, 6), (6, 12), (12, 24)]):  # 按预报时长分组
        R, X = [], []
        for d_ in days:
            base = base_fn(d_)                        # Q2 主方案点预测
            dv0 = pv_energy(D, pvfc[d_], d_, 0) - base_fn(d_, 'pv')
            m = (LEAD_H >= a) & (LEAD_H < b)          # 预报时长掩码
            R.append((D['net'][d_] - base)[m]);  X.append(dv0[m])
        r, x = np.concatenate(R), np.concatenate(X)
        if x @ x > 1e-9:
            k0[g] = np.clip(-(r @ x) / (x @ x), 0.0, 1.0)
    return k0

def adjust_lp(price, N_hat, b_plan, E_tau, E0):
    """剩余 H 区间重定合同 b'（口径B：对计划净额收违约附加费）。
    min  p.b' + 0.5*p*(u+v) + 5*p*e
    s.t. E_{t+1}=E_t+eta_c*c-d/eta_d;  b'+d+e = N_hat+c+w
         u >= b_plan-b',  v >= b'-b_plan;  E_0=E_tau;  E_H=E0 """
    H = len(N_hat)
    # 变量序 [b', c, d, e, w, u, v, E(0..H)]，按上式组装 cobj/A_eq/A_ub/bounds
    res = linprog(cobj, A_ub=Aub, b_ub=bub, A_eq=Aeq, b_eq=beq,
                  bounds=bounds, method="highs")
    return res.x[:H]

def exec_segment(b_cur, net_d, E, t0, t1):   # must/max 尽限执行
    for t in range(t0, t1):
        net = b_cur[t] - net_d[t]            # 合同交付与实际净负载之差
        if net >= 0:                         # 富余 -> 充电，装不下则弃置
            c = min(net, P_MAX, max(0, (E_MAX - E) / ETA_C))
        else:                                # 缺口 -> 放电，放不出则紧急购电
            d_ = min(-net, P_MAX, max(0, (E - E_MIN) * ETA_D))
        E = E + ETA_C * c - d_ / ETA_D
    return E                                 # 段末 SOC 传入下一段（跨段连续）
