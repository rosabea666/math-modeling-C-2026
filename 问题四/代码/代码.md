from pathlib import Path

import numpy as np

import pandas as pd

from scipy.optimize import linprog

T, DT = 144, 1 / 6            # 日区间数、区间长度(h)

ETA = 0.9                 # 充/放电效率

E_MIN, E_MAX, E_INIT = 1200., 10800., 6000.

P_MAX = 5000 * DT             # 单区间最大充/放电量(kWh)

ROOT = Path(__file__).resolve().parent.parent

\# ---------------- 数据 ----------------

a1 = pd.read_excel(ROOT / 'data/附件1.xlsx')

tou = a1.iloc[:, 1].to_numpy(float)            # 附件1 分时电价

net_ref = (a1.iloc[:, 2] - a1.iloc[:, 3]).to_numpy(float) * DT  # 代表日净负载

xl = pd.ExcelFile(ROOT / 'data/附件2.xlsx')

dates = pd.to_datetime(pd.read_excel(xl, 0).iloc[:, 0]).to_numpy()

net = (pd.read_excel(xl, 0).iloc[:, 1:145].to_numpy(float)

​    \- pd.read_excel(xl, 1).iloc[:, 1:145].to_numpy(float)) * DT  # 实测净负载

prt = pd.read_excel(ROOT / 'data/附件4.xlsx').iloc[:, 1:145].to_numpy(float)

Fpt = pd.read_csv(ROOT / 'results/q4_v3/F_pt.csv', header=None).to_numpy()

\# ---------------- 决策层 ----------------

def plan_lp(p, N, E0):

  """日前计划 LP：min Σ p·g，g+d=N+c+w，SOC 递推，末端 E=E0（48h 循环）。"""

  H = len(N); t = np.arange(H); n = 5 * H + 1

  Aeq = np.zeros((2 * H + 2, n)); beq = np.zeros(2 * H + 2)

  Aeq[t, 4*H+1+t] = 1; Aeq[t, 4*H+t] = -1

  Aeq[t, H+t] = -ETA; Aeq[t, 2*H+t] = 1 / ETA      # SOC 递推

  Aeq[H+t, t] = 1; Aeq[H+t, 2*H+t] = 1

  Aeq[H+t, H+t] = -1; Aeq[H+t, 3*H+t] = -1; beq[H+t] = N  # 功率平衡

  Aeq[2*H, 4*H] = 1; beq[2*H] = E0           # 起始电量

  Aeq[2*H+1, 5*H] = 1; beq[2*H+1] = E0         # 时域末端=起始

  x = linprog(np.r_[p, np.zeros(n-H)], A_eq=Aeq, b_eq=beq, method='highs',

​        bounds=[(0, None)]*H + [(0, P_MAX)]*(2*H) + [(0, None)]*H

​        \+ [(E_MIN, E_MAX)]*(H+1)).x

  return x[:H]

def adjust_lp(p, N, b, E0, E_end, p2, N2):

  """调整 LP：当日余段(长 h)合同 b' 带 0.5p|Δ| 附加费；次日(144)暂拟无附加费。"""

  H = len(N) + 144; t = np.arange(H); h = len(N); n = 4*H + 1 + 2*h

  iu, iv = 4*H + 1, 4*H + 1 + h

  obj = np.r_[p, p2, np.zeros(3*H + 1), .5*p, .5*p]

  Aeq = np.zeros((H + 2, n)); beq = np.r_[np.zeros(H), E0, E_end]

  Aeq[t, 3*H+1+t] = 1; Aeq[t, 3*H+t] = -1

  Aeq[t, H+t] = -ETA; Aeq[t, 2*H+t] = 1 / ETA      # SOC 递推

  Aeq[H, 3*H] = 1; Aeq[H+1, 4*H] = 1          # 起止电量

  Aub = np.zeros((H + 2*h, n)); bub = np.r_[-np.r_[N, N2], b, -b]

  Aub[t, t] = -1; Aub[t, H+t] = 1; Aub[t, 2*H+t] = -1  # g+d-w <= N

  j = np.arange(h)

  Aub[H+j, j] = 1; Aub[H+j, iu+j] = -1         # b'-b <= u

  Aub[H+h+j, j] = -1; Aub[H+h+j, iv+j] = -1       # b-b' <= v

  x = linprog(obj, A_ub=Aub, b_ub=bub, A_eq=Aeq, b_eq=beq, method='highs',

​        bounds=[(0, None)]*H + [(0, P_MAX)]*(2*H)

​        \+ [(E_MIN, E_MAX)]*(H+1) + [(0, None)]*(2*h)).x

  return x[:h]

def exec_seg(bf, nd_d, E, t0, t1):

  """实时执行：计划盈余尽限充电、缺口尽限放电，剩余缺口为紧急购电。"""

  em = np.zeros(t1 - t0)

  for k, t in enumerate(range(t0, t1)):

​    r = bf[t] - nd_d[t]

​    if r >= 0:

​      c = min(r, P_MAX, max(0, (E_MAX - E) / ETA)); E += ETA * c

​    else:

​      d = min(-r, P_MAX, max(0, (E - E_MIN) * ETA))

​      em[k] = -r - d; E -= d / ETA

  return em, E

def margin(d, beta):

  """近 7 日净负载的中心化 β 分位数裕量（Q2 同款，W=7 逐月走查选定）。"""

  h = net[max(0, d-7):d]

  return np.quantile(h - np.median(h, axis=0), np.clip(beta, .05, .99), axis=0)

\# ---------------- 全年回放 ----------------

def run(adjust, lam, delta):

  E = E_INIT

  for d in range(31):                  # 1 月预热（代表日计划+日循环）

​    b = plan_lp(tou, net_ref, E); _, E = exec_seg(b, net[d], E, 0, T)

  cost = 0.

  for d in range(31, len(dates)):            # 评价期 2/1-12/31

​    month = pd.Timestamp(dates[d]).month

​    beta = (.70 if month in (2, 3) else .75) + delta

​    w = np.clip((tou / tou.mean()) ** lam, .5, 2)   # PRM 权重

​    N = Fpt[d] + w * margin(d, beta)         # 需求代理

​    E0 = E

​    b = plan_lp(np.r_[tou, tou], np.r_[N, N], E)[:T]  # 48h：次日持久性平铺

​    bf = b.copy()

​    for tau in (0, 6, 12, 18):

​      t0, t1 = tau * 6, tau * 6 + 36

​      if tau and adjust:              # 日内三次调整(同 48h 结构)

​        bf[t0:] = adjust_lp(tou[t0:], N[t0:], b[t0:], E, E0, tou, N)

​      em, E = exec_seg(bf, net[d], E, t0, t1)

​      p = prt[d, t0:t1]

​      cost += 5 * p @ em              # 紧急购电费

​    cost += prt[d] @ bf + .5 * prt[d] @ np.abs(bf - b) * adjust

  return cost / 1e4
